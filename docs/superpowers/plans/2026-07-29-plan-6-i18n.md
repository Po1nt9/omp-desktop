# Plan 6: i18n — Dialog Localization, v1 Error Keys, Completeness Script

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Close the remaining i18n gaps: localize the two hardcoded English save-dialog titles in `commands.rs`, fix two brand-violating fallback filenames, surface v1 protocol error `messageKey`s through the frontend catalog so v1 errors render localized, and add a standalone completeness-check script that parallels `check-brand-policy.mjs`.

**Architecture:** The codebase already has a hand-rolled i18n system (`src/i18n/`) covering `en`, `zh`, `zh-TW` with parity tests, plus a Rust-side `tray_i18n.rs` for native menu labels. Plan 6 extends the Rust-side pattern with a new `dialog_i18n.rs` (mirroring `tray_i18n.rs`), adds 9 v1 error `messageKey` strings to all three frontend locales, and introduces `scripts/check-i18n-completeness.mjs` as a standalone Node script (no vitest dependency) using `typescript`'s `transpileModule` to load the `.ts` catalog at runtime.

**Tech Stack:** Rust (Tauri side: `dialog_i18n.rs`), TypeScript (frontend catalog + standalone check script), Node + `typescript` package (already in devDeps) for transpile-on-the-fly.

**Locale ID convention:** The codebase uses `"zh"` (not `"zh-CN"`) as the canonical Simplified Chinese id; `resolveLocale()` and `Locale::parse()` already accept `zh-CN`/`zh-Hans`/`zh_cn` aliases. Plan 6 keeps `"zh"` as canonical and updates the roadmap wording to match reality (no settings migration needed — aliases cover it).

```rust
    crate::tray_i18n::app_locale()

    /// Title for the support bundle save dialog (commands::export_support_bundle).
    pub support_bundle_title: &'static str,
    /// Title for the session diagnostic bundle save dialog (commands::export_session_bundle).
    pub session_bundle_title: &'static str,

    support_bundle_title: "Save support bundle",
    session_bundle_title: "Save session diagnostic bundle",

    support_bundle_title: "保存支持诊断包",
    session_bundle_title: "保存会话诊断包",

    session_bundle_title: "儲存工作階段診斷包",

    match locale {
        Locale::En => &EN,
        Locale::Zh => &ZH,
```

```js
// Standalone i18n completeness checker.
// Loads src/i18n/messages.ts via typescript.transpileModule (no vitest dependency),
// asserts key parity + non-empty values across all locales, and exits 1 on any gap.
// Parallels check-brand-policy.mjs in shape and exit-code convention.

import fs from "node:fs";
import path from "node:path";
import { fileURLToPath } from "node:url";
import ts from "typescript";

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), "..");

function loadTsModule(filePath) {
  const source = fs.readFileSync(filePath, "utf8");
  const js = ts.transpileModule(source, {
    compilerOptions: {
      module: ts.ModuleKind.ESNext,
      target: ts.ScriptTarget.ES2022,
    },
  }).outputText;
  const dataUrl = `data:text/javascript;base64,${Buffer.from(js).toString("base64")}`;
  return import(dataUrl);
}

export function checkCatalog(messages) {
  const locales = Object.keys(messages);
  const violations = [];

  if (locales.length === 0) {
    violations.push("no locales found in messages export");
    return violations;
  }

  const enKeys = new Set(Object.keys(messages.en ?? {}));
  if (enKeys.size === 0) {
    violations.push("no keys found in messages.en (source locale is empty)");
    return violations;
  }

  for (const loc of locales) {
    const table = messages[loc];
    if (!table || typeof table !== "object") {
      violations.push(`${loc}: locale table is missing or not an object`);
      continue;
    }
    const keys = Object.keys(table);

    // Missing keys
    for (const k of enKeys) {
      if (!(k in table)) {
        violations.push(`${loc}: missing key "${k}"`);
        continue;
      }
      const v = table[k];
      if (typeof v !== "string") {
        violations.push(`${loc}.${k}: value is not a string (got ${typeof v})`);
      } else if (v.trim().length === 0) {
        violations.push(`${loc}.${k}: value is empty or whitespace-only`);
      }
    }

    // Extra keys
    for (const k of keys) {
      if (!enKeys.has(k)) {
        violations.push(`${loc}: extra key "${k}" not present in en`);
      }
    }
  }

  return violations;
}

async function main() {
  const messagesMod = await loadTsModule(path.join(root, "src/i18n/messages.ts"));
  const messages = messagesMod.messages;
  if (!messages) {
    console.error("check-i18n: failed to load messages export from src/i18n/messages.ts");
    process.exitCode = 1;
    return;
  }

  const violations = checkCatalog(messages);
  if (violations.length) {
    for (const v of violations) console.error(`check-i18n: ${v}`);
    console.error(`check-i18n: ${violations.length} violation(s)`);
    process.exitCode = 1;
    return;
  }

  const locales = Object.keys(messages);
  const enKeys = Object.keys(messages.en).length;
  console.log(`check-i18n: OK (${locales.length} locales, ${enKeys} keys each)`);
}

if (process.argv[1] === fileURLToPath(import.meta.url)) {
  await main();
}
```

- [ ] **Step 2: Write the test fixture**

Create `scripts/check-i18n-completeness.test.mjs`:

```js
import { describe, expect, it } from "vitest";
import { checkCatalog } from "./check-i18n-completeness.mjs";

describe("check-i18n-completeness", () => {
  it("passes for a clean catalog", () => {
    const messages = {
      en: { "a.ok": "OK", "b.title": "Title" },
      zh: { "a.ok": "好的", "b.title": "标题" },
    };
    expect(checkCatalog(messages)).toEqual([]);
  });

  it("flags a missing key", () => {
    const messages = {
      en: { "a.ok": "OK", "b.title": "Title" },
      zh: { "a.ok": "好的" },
    };
    const v = checkCatalog(messages);
    expect(v).toContain('zh: missing key "b.title"');
  });

  it("flags an empty value", () => {
    const messages = {
      en: { "a.ok": "OK" },
      zh: { "a.ok": "   " },
    };
    const v = checkCatalog(messages);
    expect(v.some((s) => s.startsWith("zh.a.ok: value is empty"))).toBe(true);
  });

  it("flags an extra key not in en", () => {
    const messages = {
      en: { "a.ok": "OK" },
      zh: { "a.ok": "好的", "b.extra": "额外" },
    };
    const v = checkCatalog(messages);
    expect(v).toContain('zh: extra key "b.extra" not present in en');
  });

  it("fails when en is empty", () => {
    const messages = { en: {}, zh: {} };
    const v = checkCatalog(messages);
    expect(v.some((s) => s.includes("source locale is empty"))).toBe(true);
  });
});
```
